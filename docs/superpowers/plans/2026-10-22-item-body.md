- [ ] **Step 2: Implement**

```go
package domain

import (
	"fmt"
	"strings"
	"unicode/utf8"
)

const MaxBodyRunes = 100_000

// NormalizeBody trims space; empty → nil; over MaxBodyRunes → error.
func NormalizeBody(s string) (*string, error) {
	t := strings.TrimSpace(s)
	if t == "" {
		return nil, nil
	}
	if utf8.RuneCountInString(t) > MaxBodyRunes {
		return nil, fmt.Errorf("body exceeds %d characters", MaxBodyRunes)
	}
	return &t, nil
}
```

- [ ] **Step 1: Write migration**

```sql
alter table public.items
  add column if not exists body text;

alter table public.items
  drop constraint if exists items_body_rune_length_check;

-- Defense in depth: character_length ≈ Unicode code points for BMP; CLI/gateway use rune count.
alter table public.items
  add constraint items_body_char_length_check
  check (body is null or char_length(body) <= 100000);

comment on column public.items.body is 'Optional large text payload (create-time); title remains the short headline.';
```
